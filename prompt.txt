const API_URL = import.meta.env.VITE_APP_BASE_URL;

const resolveAccessToken = (options?: { debug?: boolean }): string => {
  const debug = options?.debug ?? false;
  const rawValue = localStorage.getItem("access_token");

  if (!rawValue) {
    throw new Error("Token non trouvé");
  }

  if (debug) {
    console.log("🔍 resolveAccessToken :: valeur brute (type):", typeof rawValue);
    console.log("🔍 resolveAccessToken :: valeur brute (début):", rawValue.substring(0, 100));
  }

  try {
    const parsed = JSON.parse(rawValue);

    if (debug) {
      console.log("🔍 resolveAccessToken :: valeur parsée (type):", typeof parsed);
      console.log("🔍 resolveAccessToken :: valeur parsée:", parsed);
    }

    if (typeof parsed === "string" && parsed.trim().length > 0) {
      return parsed.trim();
    }

    if (parsed && typeof parsed === "object") {
      const candidate =
        (parsed as { access_token?: unknown; token?: unknown }).access_token ??
        (parsed as { access_token?: unknown; token?: unknown }).token;

      if (typeof candidate === "string" && candidate.trim().length > 0) {
        return candidate.trim();
      }

      if (debug) {
        console.warn("🔍 resolveAccessToken :: objet sans access_token valide:", parsed);
      }
    }
  } catch (parseError) {
    if (debug) {
      console.warn("🔍 resolveAccessToken :: JSON.parse a échoué, utilisation directe");
    }

    if (rawValue.trim().length > 0) {
      return rawValue.trim();
    }

    throw new Error("Le token brut est vide");
  }

  throw new Error("Impossible d'extraire un access_token valide");
};

export const loginUser = async (credentials: any) => {
  try {
    // Vérifier si un token existe déjà dans localStorage (partagé entre tous les onglets)
    const existingToken = localStorage.getItem("access_token");
    if (existingToken) {
      // Si un token existe, empêcher la connexion avec un autre compte
      return {
        data: null,
        message: "Une session est déjà active",
        isActivated: false,
        sessionLocked: true,
      };
    }

    let intersection: any[] = [];
    let isActivated = false;
    const res = await axios.post(
      `${API_URL}/serviceauth/auth/login`,
      credentials
    );
    const roles = (await getAllRolesForLogin()).roles.map((role: any) => role.role_id);
    const decoded = jwtDecode(res.data.access_token);
    console.log("DECODED: ",decoded)
    const userRoles = (decoded as any).roles.map((role: any) => role.role_id);
    intersection = userRoles.filter((value: any) => roles.includes(value));
    if (intersection.length >= 2) {
      isActivated = true;
    }
    
    if (userRoles.some((el: number) => roles.includes(el))) {
      return { data: res.data, message: "Connecté avec succès", isActivated };
    } else {
      return {
        data: null,
        message:
          "Vous n'avez pas les droits d'accès, veuillez contacter un administrateur.",
      };
    }
  } catch (error) {
    console.error("Login error:", error);
    return { data: null, message: "Veuillez vérifier vos identifiants." };
  }
};

export const registerUser = async (formData: FormData) => {
  try {
    const res = await axios.post(
      `${API_URL}/serviceauth/users/register-with-citizen-short`,
      formData,
      {
        headers: { "Content-Type": "multipart/form-data" },
      }
    );
    return res.data;
  } catch (error: any) {
    console.error("Register error:", error.response?.data || error.message);
    throw error;
  }
};

export interface UpdateProfileData {
  user_pseudo?: string;
  user_email?: string;
  user_phone?: string;
  user_password?: string;
}

export const updateProfile = async (profileData: UpdateProfileData) => {
  try {
    const token = resolveAccessToken();

    const res = await axios.patch(
      `${API_URL}/serviceauth/auth/profile`,
      profileData,
      {
        headers: {
          Authorization: `Bearer ${token}`,
          "Content-Type": "application/json",
        },
      }
    );
    return res.data;
  } catch (error: any) {
    console.error("Update profile error:", error.response?.data || error.message);
    throw error;
  }
};

export const resetUserPassword = async (userId: string, newPassword: string) => {
  try {
    const token = resolveAccessToken();

    const res = await axios.post(
      `${API_URL}/serviceauth/auth/change-password-via-admin`,

/\*\*

 \* Récupère les informations d'un citoyen à partir de l'ID utilisateur

 \* Effectue d'abord un appel à /serviceauth/users/{id} pour récupérer l'id_citizen,

 \* puis un appel à /servicecitoyen/citizens/getCitizenById/{id_citizen} pour récupérer les données du citoyen

 \* @param userId - L'ID de l'utilisateur

 \* @returns Les informations du citoyen ou null si non trouvé

 \*/

export const getCitizenByUserId = async (userId: string): Promise\<Citizen | null> => {

  try {

    // 1. Récupérer les informations utilisateur pour obtenir l'id_citizen

    const userRes = await axios.get(\`${API_URL}/serviceauth/users/${userId}\`);

    if (!userRes.data || !userRes.data.id_citizen) {

      console.warn(\`Utilisateur ${userId} n'a pas d'id_citizen associé\`);

      return null;

    }

    const idCitizen = userRes.data.id_citizen;

    // 2. Récupérer les informations du citoyen

    const citizenRes = await axios.get\<Citizen>(\`${API_URL}/servicecitoyen/citizens/getCitizenById/${idCitizen}\`);

    return citizenRes.status === 200 ? citizenRes.data : null;

  } catch (error: any) {

    if (error?.response?.status === 404) {

      console.warn(\`Citoyen non trouvé pour l'utilisateur ${userId}\`);

    } else {

      console.error(\`Erreur lors de la récupération du citoyen pour l'utilisateur ${userId}:\`, error);

    }

    return null;

  }

};

/\*\*

 \* Récupère l'URL du preview de l'avatar d'un utilisateur

 \* Convertit l'URL de [https://gateway.tsirylab.com/serviceupload/file/hello/xxx.jpg](https://gateway.tsirylab.com/serviceupload/file/hello/xxx.jpg)

 \* en [https://gateway.tsirylab.com/serviceupload/file/preview/hello%2Fxxx.jpg](https://gateway.tsirylab.com/serviceupload/file/preview/hello%2Fxxx.jpg)

 \* @param citizen_photo - L'URL complète de la photo du citoyen

 \* @returns L'URL du preview ou null si pas de photo

 \*/

export const getCitizenAvatarPreview = (citizen_photo: string | null | undefined): string | null => {

  if (!citizen_photo) return null;

  // Extraire le chemin après "/file/"

  const fileMatch = citizen_photo.match(/**\\/**file**\\/**(.+)$/);

  if (!fileMatch) return citizen_photo; // Si pas de match, retourner l'original

  const filePath = fileMatch[1]; // ex: "hello/1759607089733-f0b55428-2b5e-4269-a48c-06c9819885f0.jpg"

  const encodedPath = filePath.replace(/**\\/**/g, '%2F'); // ex: "hello%2F1759607089733-f0b55428-2b5e-4269-a48c-06c9819885f0.jpg"

  // Reconstruire l'URL avec /file/preview/

  const baseUrl = citizen_photo.substring(0, citizen_photo.indexOf('/file/'));

  return \`${baseUrl}/file/preview/${encodedPath}\`;

};

/\*\*

 \* Récupère l'URL de l'avatar avec une image par défaut en fallback

 \* @param citizen_photo - L'URL de la photo du citoyen

 \* @param defaultAvatar - URL de l'image par défaut (optionnel)

 \* @param usePreview - Utiliser la version preview (true) ou originale (false)

 \* @returns L'URL de l'avatar ou l'image par défaut

 \*/

export const getCitizenAvatarWithFallback = (

  citizen_photo: string | null | undefined,

  defaultAvatar: string = '/public/logo.png',

  usePreview: boolean = true

): string => {

  if (!citizen_photo) return defaultAvatar;

  return usePreview ? (getCitizenAvatarPreview(citizen_photo) || defaultAvatar) : (getCitizenAvatar(citizen_photo) || defaultAvatar);

};

const extractArray = (payload: any): any[] | null => {
  if (Array.isArray(payload)) return payload;
  if (payload && typeof payload === "object") {
    if (Array.isArray(payload.data)) return payload.data;
    if (payload.data && typeof payload.data === "object" && Array.isArray(payload.data.data)) {
      return payload.data.data;
    }
    if (Array.isArray(payload.results)) return payload.results;
    if (Array.isArray(payload.items)) return payload.items;
  }
  return null;
};

const fetchTerritoryList = async (path: string, contextLabel: string): Promise<any[] | null> => {

    if (import.meta.env.DEV) {
      console.debug(`[territoryService] ${contextLabel} chargés`, {
        count: data.length,
        path,
      });
    }
    return data;
  } catch (error) {
    console.error(`Erreur lors de la récupération ${contextLabel}:`, error);
    return null;
  }
};
